import type { ReactNode } from "react";
import { RestokTheme } from "@/components/ui/theme";
import { BrandMark } from "@/components/ui/brand-mark";
import { HouseSwitcher } from "./house-switcher";
import { Navigation, type AppScreen } from "./navigation";
import styles from "./app-shell.module.css";

type AppShellProps = {
  screen: AppScreen;
  onChange: (screen: AppScreen) => void;
  houseName: string;
  onHouseOpen: () => void;
  profile: ReactNode;
  feedback?: ReactNode;
  notices: ReactNode;
  noticeTitle?: string;
  children: ReactNode;
  context?: ReactNode;
  contextLayout?: "responsive" | "desktop-only";
  dock?: ReactNode;
  overlays: ReactNode;
  keyboardOpen: boolean;
};

export function AppShell({
  screen,
  onChange,
  houseName,
  onHouseOpen,
  profile,
  feedback,
  notices,
  noticeTitle,
  children,
  context,
  contextLayout = "responsive",
  dock,
  overlays,
  keyboardOpen,
}: AppShellProps) {
  const hasContext = context !== undefined && context !== null && context !== false;

  return (
    <RestokTheme className={styles.shell}>
      <a
        href="#restok-main"
        className={styles.skip}
        onClick={(event) => {
          event.preventDefault();
          document.getElementById("restok-main")?.focus();
        }}
      >
        Ir para o conteúdo
      </a>
      <div className={styles.frame}>
        <aside data-app-sidebar className={styles.sidebar}>
          <BrandMark />
          <HouseSwitcher name={houseName} onOpen={onHouseOpen} />
          <Navigation screen={screen} onChange={onChange} />
        </aside>
        <main
          id="restok-main"
          data-app-main
          data-dialog-return-focus
          data-has-dock={Boolean(dock)}
          tabIndex={-1}
          className={styles.main}
        >
          <header data-app-header className={styles.header}>
            <div className={styles.mobileHouse}>
              <HouseSwitcher name={houseName} onOpen={onHouseOpen} />
            </div>
            <span className={styles.pageContext}>
              {houseName}
            </span>
            <div className={styles.accountActions}>
              {feedback}
              {profile}
            </div>
          </header>
          {noticeTitle ? (
            <details className={styles.notices}>
              <summary>
                <span role="status">{noticeTitle}</span>
              </summary>
              {notices}
            </details>
          ) : null}
          <div
            className={styles.workspace}
            data-has-context={hasContext ? "true" : undefined}
            data-context-layout={hasContext && contextLayout === "desktop-only" ? "desktop-only" : undefined}
          >
            <div data-app-work className={styles.content}>{children}</div>
            {hasContext ? (
              <aside data-app-context className={styles.context} aria-label="Contexto da tela">
                {context}
              </aside>
            ) : null}
          </div>
        </main>
      </div>
      {!keyboardOpen ? <div className={styles.navigation}><Navigation screen={screen} onChange={onChange} /></div> : null}
      {dock ? <div data-shopping-dock className={styles.dock}>{dock}</div> : null}
      {overlays}
    </RestokTheme>
  );
}
